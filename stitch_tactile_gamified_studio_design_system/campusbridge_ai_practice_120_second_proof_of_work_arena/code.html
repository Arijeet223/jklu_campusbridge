# Redis SETNX with transaction lock & DB unique idempotency key
async def handle_razorpay_webhook(event: WebhookEvent, db: AsyncSession):
    # 1. Acquire distributed lock with 60-second TTL
    lock_acquired = await redis.set(f"lock:pay:{event.id}", "1", nx=True, ex=60)
    if not lock_acquired:
        return {"status": "ignored", "reason": "duplicate_webhook"}
    
    # 2. Database unique constraint idempotency check
    existing_txn = await db.execute(select(Payment).where(Payment.gateway_id == event.id))
    if existing_txn.scalar_one_or_none():
        return {"status": "already_processed"}
    
    # 3. Commit ledger state
    await db.commit()
